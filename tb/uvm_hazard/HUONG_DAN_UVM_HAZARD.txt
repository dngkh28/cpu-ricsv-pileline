---------------------------------------------------------------------
1. HIỂU hazard_detector TRƯỚC KHI KIỂM TRA
---------------------------------------------------------------------
hazard_detector là khối TỔ HỢP (không clock, không trạng thái), nằm trong
ex_stage. Nó xử lý hai loại hazard của pipeline:

  Load-use: lệnh ở EX là lệnh load, mà lệnh ở ID cần chính thanh ghi đó.
            Kết quả load chưa có (chỉ có sau MEM) nên phải dừng 1 chu kỳ.
  Nhảy:     lệnh ở EX là nhảy/rẽ nhánh thành công, hai lệnh đã nạp sai phải
            bỏ đi.

Cổng vào / ra:
  rs1_id, rs2_id   [4:0]  nguồn của lệnh đang ở ID
  rd_ex            [4:0]  đích của lệnh đang ở EX
  is_load_ex              lệnh ở EX là load
  pc_sel           [1:0]  00 pc+4, 01 nhảy/rẽ nhánh, 10 JALR (11 không dùng)
  pc_en, ifid_en          0 = giữ PC / giữ IF/ID (stall)
  ifid_flush              1 = xóa IF/ID
  idex_flush              1 = chèn bubble vào ID/EX

ĐẶC TẢ (dùng để viết mô hình tham chiếu, KHÔNG nhìn code RTL):
  hazard = is_load_ex VÀ rd_ex khác x0 VÀ (rd_ex trùng rs1_id HOẶC rs2_id)
  taken  = pc_sel khác 00
  pc_en = ifid_en = NOT hazard
  ifid_flush      = taken
  idex_flush      = hazard HOẶC taken

Không gian đầu vào: 5+5+5+1+2 = 18 bit, quá lớn để thử hết bằng tay, nên dùng
test ngẫu nhiên có thiên lệch. Các tình huống PHẢI thử:

  | Tình huống                                   | Mong đợi                      |
  | load, rd khác 0, trùng rs1 (không trùng rs2) | stall, bubble                 |
  | load, rd khác 0, trùng rs2 (không trùng rs1) | stall, bubble                 |
  | load, rd khác 0, trùng cả hai                | stall, bubble                 |
  | load, rd = x0, có trùng                      | KHÔNG stall                   |
  | không phải load, có trùng                    | KHÔNG stall                   |
  | pc_sel = 01 hoặc 10                          | flush IF/ID và ID/EX          |
  | stall và nhảy cùng lúc (xem lưu ý dưới)      | theo công thức                |

Lưu ý: trong CPU thật, "load ở EX" và "nhảy thành công ở EX" không thể xảy ra
cùng lúc (một lệnh không vừa là load vừa là nhảy). Nhưng ta kiểm tra khối
ĐỘC LẬP nên vẫn thử tổ hợp đó: công thức vẫn xác định rõ kết quả.

  // Transaction: một tình huống đầu vào của hazard_detector và 4 đầu ra thu được.

                       rs1_id, rs2_id, rd_ex, is_load_ex, pc_sel, pc_en, ifid_en, ifid_flush, idex_flush);
    endfunction
  endclass

BƯỚC 5: hazard_seq.sv  (SEQUENCE)
---------------------------------------------------------------------
Sinh n tình huống. Hai số 5 bit ngẫu nhiên chỉ trùng nhau 1/32 lần, nên nếu
chọn hoàn toàn ngẫu nhiên thì rất hiếm khi gặp load-use. Vì vậy ta CHỌN CÓ
THIÊN LỆCH: 25% rd_ex = 0, 25% rs1_id = rd_ex, 25% rs2_id = rd_ex.

hazard_seq.sv:

  // Sequence: sinh n tình huống. Vì không có rand/constraint nên chọn số có thiên lệch:
  // - 25% rd_ex = 0 (thanh ghi x0, không gây hazard)
  // - 25% rs1_id = rd_ex, 25% rs2_id = rd_ex (để hay gặp trường hợp trùng)
  // - pc_sel chỉ lấy 0, 1, 2 (giá trị 3 không dùng trong thiết kế)
  class hazard_seq extends uvm_sequence#(hazard_item);
    `uvm_object_utils(hazard_seq)

    int n = 200;

    function new(string name = "hazard_seq");
      super.new(name);
    endfunction

    task body();
      repeat (n) begin
        hazard_item it = hazard_item::type_id::create("it");
        start_item(it);
        it.rd_ex       = ($urandom_range(3, 0) == 0) ? 5'd0 : $urandom;
        it.rs1_id      = ($urandom_range(3, 0) == 0) ? it.rd_ex : $urandom;
        it.rs2_id      = ($urandom_range(3, 0) == 0) ? it.rd_ex : $urandom;
        it.is_load_ex  = $urandom_range(1, 0);
        it.pc_sel      = $urandom_range(2, 0);
        finish_item(it);
      end
    endtask
  endclass

Thêm vào hazard_pkg.sv:  `include "hazard_seq.sv"

BƯỚC 6: hazard_driver.sv  (DRIVER)
---------------------------------------------------------------------
Giống driver của alu: mỗi item hai chu kỳ, chu kỳ 1 đặt dữ liệu và valid=1,
chu kỳ 2 hạ valid=0 rồi item_done. Khác duy nhất: gán 5 tín hiệu đầu vào.

hazard_driver.sv:

  // Driver: đặt đầu vào lên dây. Mỗi item hai chu kỳ: chu kỳ 1 đặt dữ liệu và valid=1,
  // chu kỳ 2 hạ valid=0 rồi item_done.
  class hazard_driver extends uvm_driver#(hazard_item);
    `uvm_component_utils(hazard_driver)

    virtual hazard_if vif;

    function new(string name, uvm_component parent);
      super.new(name, parent);
    endfunction

    function void build_phase(uvm_phase phase);
      super.build_phase(phase);
      if (!uvm_config_db#(virtual hazard_if)::get(this, "", "vif", vif))
        `uvm_fatal("DRV", "khong lay duoc vif tu config_db")
    endfunction

    task run_phase(uvm_phase phase);
      vif.valid <= 0;
      forever begin
        seq_item_port.get_next_item(req);
        @(posedge vif.clk);
        vif.rs1_id     <= req.rs1_id;
        vif.rs2_id     <= req.rs2_id;
        vif.rd_ex      <= req.rd_ex;
        vif.is_load_ex <= req.is_load_ex;
        vif.pc_sel     <= req.pc_sel;
        vif.valid      <= 1;
        @(posedge vif.clk);
        vif.valid      <= 0;
        seq_item_port.item_done();
      end
    endtask
  endclass

Thêm vào hazard_pkg.sv:  `include "hazard_driver.sv"

BƯỚC 7: hazard_monitor.sv  (MONITOR)
---------------------------------------------------------------------
Khi valid=1, thu 5 đầu vào và 4 đầu ra thành một item MỚI, gửi qua ap.write.

hazard_monitor.sv:

  // Monitor: khi valid=1 thu cả đầu vào lẫn 4 đầu ra thành một item mới, gửi cho scoreboard.
  class hazard_monitor extends uvm_monitor;
    `uvm_component_utils(hazard_monitor)

    virtual hazard_if vif;
    uvm_analysis_port#(hazard_item) ap;

    function new(string name, uvm_component parent);
      super.new(name, parent);
    endfunction

    function void build_phase(uvm_phase phase);
      super.build_phase(phase);
      ap = new("ap", this);
      if (!uvm_config_db#(virtual hazard_if)::get(this, "", "vif", vif))
        `uvm_fatal("MON", "khong lay duoc vif tu config_db")
    endfunction

    task run_phase(uvm_phase phase);
      hazard_item it;
      forever begin
        @(posedge vif.clk);
        if (vif.valid === 1'b1) begin
          it = hazard_item::type_id::create("mon_it");
          it.rs1_id     = vif.rs1_id;
          it.rs2_id     = vif.rs2_id;
          it.rd_ex      = vif.rd_ex;
          it.is_load_ex = vif.is_load_ex;
          it.pc_sel     = vif.pc_sel;
          it.pc_en      = vif.pc_en;
          it.ifid_en    = vif.ifid_en;
          it.ifid_flush = vif.ifid_flush;
          it.idex_flush = vif.idex_flush;
          ap.write(it);
        end
      end
    endtask
  endclass

BƯỚC 8: hazard_scoreboard.sv  (SCOREBOARD)
---------------------------------------------------------------------
Tính 4 đầu ra đúng bằng mô hình viết theo ĐẶC TẢ (mục 1), so với DUT. Mô hình
KHÔNG được chép từ hazard_detector.v: viết lại từ câu chữ đặc tả, nếu không
lỗi của RTL sẽ lặp lại trong mô hình và không bị phát hiện.

Coverage đếm bằng tay: các biến đếm int. Mỗi tình huống ở bảng mục 1 có một
bộ đếm, nhờ đó cuối test bạn biết đã thử đủ chưa.

hazard_scoreboard.sv:

  // Scoreboard: tính 4 đầu ra đúng bằng mô hình tham chiếu viết theo ĐẶC TẢ, so với DUT.
  // Đặc tả:
  //   - Load-use: lệnh ở EX là lệnh load, rd khác x0, và rd trùng rs1 hoặc rs2 của lệnh ở ID
  //     -> giữ PC và IF/ID (pc_en = ifid_en = 0), chèn bubble vào ID/EX (idex_flush = 1)
  //   - Nhảy: pc_sel khác 00 -> xóa IF/ID (ifid_flush = 1) và ID/EX (idex_flush = 1)
  // Coverage đếm bằng tay (không dùng covergroup).
  class hazard_scoreboard extends uvm_scoreboard;
    `uvm_component_utils(hazard_scoreboard)

    uvm_analysis_imp#(hazard_item, hazard_scoreboard) imp;

    int pass, fail;
    // coverage
    int n_stall;            // số lần load-use
    int n_stall_rs1_only;   // trùng rs1, không trùng rs2
    int n_stall_rs2_only;   // trùng rs2, không trùng rs1
    int n_stall_both;       // trùng cả hai
    int n_rd0_match;        // load, rd = x0, có trùng rs: KHÔNG được stall
    int n_nonload_match;    // không phải load, có trùng rs: KHÔNG được stall
    int n_jump;             // số lần nhảy
    int n_stall_and_jump;   // load-use và nhảy cùng lúc
    int pc_sel_seen[bit [1:0]];

    function new(string name, uvm_component parent);
      super.new(name, parent);
    endfunction

    function void build_phase(uvm_phase phase);
      super.build_phase(phase);
      imp = new("imp", this);
    endfunction

    function void write(hazard_item t);
      bit match1, match2, hazard, taken;
      bit exp_pc_en, exp_ifid_en, exp_ifid_flush, exp_idex_flush;

      match1 = (t.rs1_id == t.rd_ex);
      match2 = (t.rs2_id == t.rd_ex);
      hazard = t.is_load_ex && (t.rd_ex != 5'd0) && (match1 || match2);
      taken  = (t.pc_sel != 2'b00);

      exp_pc_en      = !hazard;
      exp_ifid_en    = !hazard;
      exp_ifid_flush = taken;
      exp_idex_flush = hazard || taken;

      // coverage
      if (!pc_sel_seen.exists(t.pc_sel)) pc_sel_seen[t.pc_sel] = 0;
      pc_sel_seen[t.pc_sel]++;
      if (hazard) begin
        n_stall++;
        if (match1 && !match2) n_stall_rs1_only++;
        if (!match1 && match2) n_stall_rs2_only++;
        if (match1 && match2)  n_stall_both++;
      end
      if (t.is_load_ex && t.rd_ex == 5'd0 && (match1 || match2)) n_rd0_match++;
      if (!t.is_load_ex && (match1 || match2) && t.rd_ex != 5'd0) n_nonload_match++;
      if (taken) n_jump++;
      if (hazard && taken) n_stall_and_jump++;

      // so sánh
      if (t.pc_en === exp_pc_en && t.ifid_en === exp_ifid_en &&
          t.ifid_flush === exp_ifid_flush && t.idex_flush === exp_idex_flush) begin
        pass++;
      end else begin
        fail++;
        `uvm_error("SB", $sformatf("SAI: %s | mong doi pc_en=%0b ifid_en=%0b ifid_flush=%0b idex_flush=%0b",
                                   t.convert2string(), exp_pc_en, exp_ifid_en, exp_ifid_flush, exp_idex_flush))
      end
    endfunction

    function void check_phase(uvm_phase phase);
      if (pass + fail == 0) `uvm_error("SB", "scoreboard khong nhan duoc giao dich nao")
    endfunction

    function void report_phase(uvm_phase phase);
      `uvm_info("SB", $sformatf("pass=%0d fail=%0d", pass, fail), UVM_LOW)
      `uvm_info("SB", $sformatf("coverage: stall=%0d (rs1 only=%0d, rs2 only=%0d, ca hai=%0d) | rd=x0 co trung=%0d | khong phai load co trung=%0d",
                n_stall, n_stall_rs1_only, n_stall_rs2_only, n_stall_both, n_rd0_match, n_nonload_match), UVM_LOW)
      `uvm_info("SB", $sformatf("coverage: nhay=%0d | stall va nhay cung luc=%0d | pc_sel da gap=%0d/3",
                n_jump, n_stall_and_jump, pc_sel_seen.size()), UVM_LOW)
      if (fail == 0 && pass > 0) `uvm_info("SB", "*** TEST PASSED ***", UVM_NONE)
      else                       `uvm_info("SB", "*** TEST FAILED ***", UVM_NONE)
    endfunction
  endclass

      hazard_seq seq = hazard_seq::type_id::create("seq");
      phase.raise_objection(this);
      seq.n = 2000;
      seq.start(env.agt.sqr);
      repeat (3) @(posedge env.agt.drv.vif.clk);
      phase.drop_objection(this);
    endtask
  endclass
